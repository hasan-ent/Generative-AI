\textbf{2. Tokenize $\to$ vocab $\to$ IDs ($\to$ \texttt{token\_ids.csv})}
\begin{lstlisting}
from nltk.tokenize import word_tokenize
toks = word_tokenize(sent.lower())              # keep tokens with [A-Za-z]
tok.build_vocab(tokenized)                      # count, drop freq<2
# word2id = {<pad>:0, <unk>:1, the:2, you:3, ...}  (339 types)
ids = [tok.word2id.get(t, 1) for t in toks]     # e.g. [hear,me,speak]->[43,1,93]
# CSV row: sentence_id, sentence, "toks...", "ids...", length
\end{lstlisting}

\textbf{3. Bigram LM with add-1 ($\to$ \texttt{language\_model.json})}
\begin{lstlisting}
for sent in tokenized:                          # OOV already mapped to <unk>
    for i, w in enumerate(sent):
        unigram[w] += 1
        if i > 0: bigram[(sent[i-1], w)] += 1
P(w | prev) = (C(prev, w) + 1) / (C(prev) + V)  # V = 339
PPL = exp(-mean(log P))                         # train PPL = 107.55
\end{lstlisting}

\textbf{4. WordNet all-pairs similarity, keep $>0.5$ ($\to$ \texttt{wordnet\_similarity.csv})}
\begin{lstlisting}
from nltk.corpus import wordnet as wn
from itertools import combinations
# candidates: top-80 frequent, alpha, len>2, no stopwords
syn = lambda w: wn.synsets(w.lower())
def score(a, b):                                # first synset = frequent sense
    A, B = syn(a), syn(b)
    return A[0].wup_similarity(B[0]) if A and B else None  # path gives ~0!
kept = [(a, b, round(s, 4)) for a, b in combinations(words, 2)
        if (s := score(a, b)) and s > 0.5]      # 110 pairs, sort desc
\end{lstlisting}

\textbf{5. Top-5 pairs $\to$ co-occurrence embeddings ($\to$ \texttt{top5\_embeddings.csv})}
\begin{lstlisting}
targets = unique(words of top-5 pairs)  # 10 words: done,made,man,sir,...
for sent in tokenized:                          # window = 2, symmetric
    f = [t for t in sent if t in targets]
    for i, w in enumerate(f):
        for j in range(max(0,i-2), min(len(f),i+3)):
            if i != j: mat[w, f[j]] += 1
emb = mat / ||mat||                             # L2 rows -> project to 50-d
cos(a, b) = a.b / (||a|| * ||b||)               # e.g. sir-like 0.837
\end{lstlisting}
